```python
import time

class RecordingRetriever:
    """旁路记录 RetrievalHit;仅供 retrieval_hit_types_json 诊断 + fallback_reason"""

    def __init__(self, inner) -> None:
        self.inner = inner
        self.last_hits: list = []

    def __getattr__(self, name):
        return getattr(self.inner, name)

    async def search(self, *args, **kwargs):
        hits = await self.inner.search(*args, **kwargs)
        self.last_hits = hits
        return hits


class RecordingTextProvider:
    """旁路记录 raw LLM response + LLM 调用 latency"""

    def __init__(self, inner) -> None:
        self.inner = inner
        self.last_raw_response_text: str | None = None
        self.last_llm_duration_ms: int | None = None

    def __getattr__(self, name):
        return getattr(self.inner, name)

    async def complete(self, *args, **kwargs):
        start = time.perf_counter()
        result = await self.inner.complete(*args, **kwargs)
        self.last_llm_duration_ms = int((time.perf_counter() - start) * 1000)
        self.last_raw_response_text = extract_raw_text_if_available(result)
        return result


class RecordingPromptBuilder:
    """R4-P0-1 NEW:旁路捕获 ChatPromptBuilder 实际收到的 source_entries

    source_id (S1/S2/...) 由 ChatService 构造 source_entries 时分配,
    不等同于 Retriever.hits 顺序。RecordingPromptBuilder 是 citation 评分的
    唯一可靠真值源。
    """

    def __init__(self, inner) -> None:
        self.inner = inner
        self.last_source_entries: list = []
        self.last_source_table: list[dict] = []

    def __getattr__(self, name):
        return getattr(self.inner, name)

    def build_messages(self, *args, **kwargs):
        # 签名由 Stage 0 #4b-2 报告确认;假设含 source_entries 关键字或位置参数
        source_entries = extract_source_entries_from_call_if_available(args, kwargs)
        self.last_source_entries = source_entries
        self.last_source_table = derive_source_table_from_entries_if_available(source_entries)
        return self.inner.build_messages(*args, **kwargs)
```
