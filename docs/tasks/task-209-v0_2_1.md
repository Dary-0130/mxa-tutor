```python
def classify_extension(ext: str) -> Literal["allow", "skip", "deny", "other"]:
    """判定顺序锁定:skip → allow → deny → other(P0-3 修订)。
```

### `_c_source_splitter.py` 接口

```python
# features/chunking/_c_source_splitter.py
from dataclasses import dataclass
from typing import Literal

CSectionKind = Literal[
    "header_comment",
    "define_block",
    "typedef_block",
    "global_var_block",
    "function_body",
    "preprocessor_block",
]

@dataclass(frozen=True)
class CSourceSection:
    index: int
    title: str               # 段标题,如 "mdlOutputs" / "PWM defines" / "global vars"
    kind: CSectionKind
    line_start: int          # 1-indexed
    line_end: int
    code: str                # 该段原始代码(供 source_text 模板使用,不直接入 chunk)

def split_c_source(raw_code: str, max_tokens: int = 1500) -> list[CSourceSection]:
    """按 C 顶层结构切段。

    切段规则:
    1. 文件头连续注释块(/* ... */ 或 // 开头连续行)→ kind=header_comment
    2. 连续 #include / #define 块 → kind=define_block / preprocessor_block
    3. typedef 块(typedef struct {...} Name; 完整一段)→ kind=typedef_block
    4. 文件级变量声明(非 static 函数外的 float/int/Uint32 等)→ kind=global_var_block
    5. 函数定义(static\\s+\\w+\\s+\\w+\\s*\\(.*\\)\\s*\\{ 起,到匹配 } 止)→ kind=function_body
    6. 函数体本身**不再细分**;若超 max_tokens,整段保留 + 日志告警

    返回:有序 list,index 从 0 起。
    """
```
